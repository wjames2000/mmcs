# 多模型协作系统 —— 详细设计说明书

## 1. 引言

### 1.1 编写目的

本文档为「多模型圆桌会议系统」的详细设计说明书，在概要设计说明书的基础上，对每个模块的**数据结构、算法逻辑、接口协议、数据库操作、配置细节**进行精确到字段级的定义，作为编码实现的直接依据。

### 1.2 适用范围

本文档面向开发工程师，用于指导编码实现。每个模块的实现应符合本文档定义的接口签名、数据结构和业务流程。

### 1.3 术语与缩写

| 术语 | 说明 |
|------|------|
| GraphNode | Eino Graph 中的处理节点，实现 Runnable[I,O] 接口 |
| SSEClient | SSE 连接的一个订阅者 |
| Hub | 每个 Session 一个的事件广播器 |
| Asynq Task | Asynq 任务队列中的最小工作单元 |
| Provider | 模型提供商适配器，实现 ChatModel 接口 |
| State | Graph 全局状态，贯穿整个 Session 生命周期 |

```go
// cmd/api-server/main.go
func main() {
    cfg := config.Load("config.yaml")

    db := postgres.New(cfg.Database)
    rdb := redis.New(cfg.Redis)
    asynqClient := asynq.NewClient(asynq.RedisClientOpt{Addr: cfg.Redis.Addr})

    // 初始化各模块
    workspaceSvc := workspace.NewService(db)
    roleSvc := role.NewService(db)
    modelGw := model_gateway.New(cfg.ModelGateway)
    agentExecutor := agent.NewExecutor(cfg.Agent, asynqClient, modelGw)
    orchestrator := orchestrator.New(roleSvc, modelGw)
    sessionSvc := session.NewService(cfg.Session, db, rdb, orchestrator)
    taskSvc := task.NewService(db, agentExecutor)
    validationSvc := validation.NewService(db, agentExecutor)
    streamSvc := stream.NewService(rdb)
    minutesSvc := minutes.NewService(db)
    authSvc := user.NewAuthService(db, cfg.Auth)

    // 注册路由
    router := api.NewRouter(authSvc, sessionSvc, workspaceSvc, roleSvc, taskSvc, validationSvc, streamSvc, minutesSvc)
    router.Serve(cfg.Server.HTTPPort)
}
```

### 3.6 SSE 事件详细格式

```go
type SSEEvent struct {
    Type    SSEEventType `json:"type"`
    Data    any          `json:"data"`
    ID      string       `json:"id,omitempty"`
    Retry   int          `json:"retry,omitempty"` // 重连间隔 ms
}
```

| 事件类型 | 触发时机 | Data 结构 |
|---------|---------|-----------|
| `session.created` | 会话创建成功 | `{session_id, title, paradigm}` |
| `round.start` | 新轮次开始 | `{round, phase}` |
| `role.speak` | 角色发言（流式） | `{role_id, role_name, token, seq}` |
| `role.done` | 角色发言结束 | `{role_id, full_content}` |
| `round.eval` | 主持人评估结果 | `{round, should_continue, summary}` |
| `session.paused` | 会话被暂停 | `{interrupt_at, message}` |
| `session.resumed` | 会话恢复 | `{resume_at}` |
| `session.ended` | 会话结束 | `{session_id, summary_url}` |
| `task.created` | 新任务创建 | `{task_id, title, priority}` |
| `task.updated` | 任务状态变更 | `{task_id, status}` |
| `error` | 运行时错误 | `{code, message}` |

SSE 事件示例：

```
event: round.start
data: {"round":1,"phase":"brainstorm"}
id: evt_round_1

event: role.speak
data: {"role_id":"r1","role_name":"安全专家","token":"从","seq":1}
id: evt_r1_001

event: role.speak
data: {"role_id":"r1","role_name":"安全专家","token":"安全","seq":2}
id: evt_r1_002

event: role.done
data: {"role_id":"r1","full_content":"从安全角度来看，该模块存在 SQL 注入风险..."}
id: evt_r1_end

event: session.ended
data: {"session_id":"s_abc123","minutes_url":"/api/v1/sessions/s_abc123/minutes"}
id: evt_end
```

## 4. 数据库详细设计

### 4.1 DDL（完整版）

```sql
-- 001_init.up.sql

-- 扩展：向量搜索
CREATE EXTENSION IF NOT EXISTS vector;

-- ===== 用户表 =====
CREATE TABLE users (

-- ===== 工作区表 =====
CREATE TABLE workspaces (
    id          VARCHAR(64) PRIMARY KEY,
    name        VARCHAR(256) NOT NULL,
    description TEXT,
    mode        VARCHAR(32) NOT NULL DEFAULT 'standalone', -- standalone / shared
    status      VARCHAR(32) NOT NULL DEFAULT 'active',     -- active / archived
    members     TEXT[] NOT NULL DEFAULT '{}',
    creator_id  VARCHAR(64) NOT NULL REFERENCES users(id),
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_workspaces_creator ON workspaces(creator_id);
CREATE INDEX idx_workspaces_members ON workspaces USING GIN(members);

-- ===== 角色表 =====
CREATE TABLE roles (
    id            VARCHAR(64) PRIMARY KEY,
    name          VARCHAR(128) NOT NULL,
    title         VARCHAR(128) NOT NULL,
    traits        JSONB NOT NULL DEFAULT '{}',         -- {aggressiveness, optimism, creativity, detail}
    expertise     TEXT[] NOT NULL DEFAULT '{}',
    speaking_style TEXT,
    system_prompt TEXT,
    skills        TEXT[] NOT NULL DEFAULT '{}',          -- 引用的 Skill 名称列表
    default_model JSONB,                               -- {provider, model_name, parameters}
    is_global     BOOLEAN NOT NULL DEFAULT false,      -- 全局模板 vs 用户自建
    creator_id    VARCHAR(64) REFERENCES users(id),
    created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_roles_creator ON roles(creator_id);
CREATE INDEX idx_roles_global ON roles(is_global) WHERE is_global = true;
CREATE INDEX idx_roles_skills ON roles USING GIN(skills);

-- ===== 会话表 =====
CREATE TABLE sessions (
    id          VARCHAR(64) PRIMARY KEY,
    workspace_id VARCHAR(64) NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
    title       VARCHAR(256) NOT NULL,
    paradigm    VARCHAR(64) NOT NULL,          -- round_robin / court / evaluation / free_chat
    status      VARCHAR(32) NOT NULL DEFAULT 'draft',  -- draft / running / paused / ended / failed
    max_rounds  INT NOT NULL DEFAULT 10,
    round_timeout INT NOT NULL DEFAULT 300,   -- 秒
    config      JSONB NOT NULL DEFAULT '{}',   -- 扩展配置
    creator_id  VARCHAR(64) NOT NULL REFERENCES users(id),
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    started_at  TIMESTAMPTZ,
    ended_at    TIMESTAMPTZ
);
CREATE INDEX idx_sessions_workspace ON sessions(workspace_id);
CREATE INDEX idx_sessions_creator ON sessions(creator_id);
CREATE INDEX idx_sessions_status ON sessions(status);
CREATE INDEX idx_sessions_workspace_status ON sessions(workspace_id, status);

-- ===== 会话角色绑定表 =====
CREATE TABLE session_roles (
    id            VARCHAR(64) PRIMARY KEY,
    session_id    VARCHAR(64) NOT NULL REFERENCES sessions(id) ON DELETE CASCADE,
    role_id       VARCHAR(64) NOT NULL REFERENCES roles(id),
    model_override JSONB,                     -- 覆盖绑定的模型 {provider, model_name, parameters}
    sort_order    INT NOT NULL DEFAULT 0,
    UNIQUE(session_id, role_id)
);
CREATE INDEX idx_session_roles_session ON session_roles(session_id);

-- ===== 消息表 =====
CREATE TABLE messages (
    id          VARCHAR(64) PRIMARY KEY,
    session_id  VARCHAR(64) NOT NULL REFERENCES sessions(id) ON DELETE CASCADE,
    round_id    VARCHAR(64),
    role_id     VARCHAR(64),
    author_type VARCHAR(32) NOT NULL,           -- ai / user / system
    role_name   VARCHAR(128),
    content     TEXT NOT NULL,
    tokens      INT NOT NULL DEFAULT 0,
    metadata    JSONB NOT NULL DEFAULT '{}',    -- {model, latency, seq}
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_messages_session ON messages(session_id);
CREATE INDEX idx_messages_round ON messages(session_id, round_id);
CREATE INDEX idx_messages_created ON messages(session_id, created_at);

-- ===== 任务表 =====
CREATE TABLE tasks (
    id                  VARCHAR(64) PRIMARY KEY,
    session_id          VARCHAR(64) NOT NULL REFERENCES sessions(id),
    workspace_id        VARCHAR(64) NOT NULL REFERENCES workspaces(id),
    title               VARCHAR(256) NOT NULL,
    description         TEXT,
    acceptance_criteria TEXT,
    status              VARCHAR(32) NOT NULL DEFAULT 'pending',
        -- pending / in_progress / reviewing / completed / rejected
    priority            VARCHAR(16) NOT NULL DEFAULT 'medium',
        -- low / medium / high / critical
    assigned_agent      VARCHAR(64),
    assigned_by         VARCHAR(64) REFERENCES users(id),
    source_round        INT,                   -- 来源轮次
    created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    completed_at        TIMESTAMPTZ
);
CREATE INDEX idx_tasks_session ON tasks(session_id);
CREATE INDEX idx_tasks_workspace ON tasks(workspace_id);
CREATE INDEX idx_tasks_status ON tasks(status);
CREATE INDEX idx_tasks_assignee ON tasks(assigned_agent);
CREATE INDEX idx_tasks_workspace_status ON tasks(workspace_id, status);

-- ===== 验证结果表 =====
CREATE TABLE validation_results (
    id          VARCHAR(64) PRIMARY KEY,
    task_id     VARCHAR(64) NOT NULL REFERENCES tasks(id) ON DELETE CASCADE,
    validator   VARCHAR(64) NOT NULL,            -- 验证官 Agent ID 或名称
    verdict     VARCHAR(32) NOT NULL,            -- passed / needs_revision / rejected
    reason      TEXT NOT NULL,
    detail      JSONB NOT NULL DEFAULT '{}',     -- {checked_items, scores}
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(task_id)
);

-- ===== 模型配置表 =====
CREATE TABLE model_providers (
    id                VARCHAR(64) PRIMARY KEY,
    user_id           VARCHAR(64) NOT NULL REFERENCES users(id),
```

```go
        SELECT status, COUNT(*) as cnt
        FROM tasks
        WHERE workspace_id = $1
        GROUP BY status`, workspaceID)
    if err != nil {
        return nil, err

        var count int
        if err := rows.Scan(&status, &count); err != nil {

        }
        agg.ByStatus[status] = count
        agg.Total += count
    }
    return agg, nil
}
```

### 5.3 expert_speak 节点

```go
type ExpertSpeakNode struct{}

type RoundContext struct {
    Round    int
    Phase    string  // brainstorm / critique / scoring
    History  []*Message  // 当前轮之前的所有消息
    Context  *SessionContext
}

type RoundFeedback struct {
    Round    int
    Opinions []*Opinion  // 每位角色的发言，并行收集
}

type Opinion struct {
    RoleID    string
    RoleName  string
    Content   string
    Tokens    int
    Latency   time.Duration
}

func (n *ExpertSpeakNode) Invoke(ctx context.Context, rc *RoundContext) (*RoundFeedback, error) {
    feedback := &RoundFeedback{Round: rc.Round}
    type result struct {
        index int
        op    *Opinion
        err   error
    }
    ch := make(chan result, len(rc.Context.Roles))

    // 并行调用各角色 ChatModel
    for i, role := range rc.Context.Roles {
        i, role := i, role
        go func() {
            start := time.Now()
            messages, _ := role.Template.Format(map[string]any{
                "round":    rc.Round,
                "phase":    rc.Phase,
                "history":  rc.History,
            })
            resp, err := role.Model.Generate(ctx, messages)
            if err != nil {
                ch <- result{i, nil, err}
                return
            }
            ch <- result{i, &Opinion{
                RoleID:   role.RoleID,
                RoleName: role.Name,
                Content:  resp.Content,
                Tokens:   resp.Tokens,
                Latency:  time.Since(start),
            }, nil}
        }()
    }

    // 收集结果
    feedback.Opinions = make([]*Opinion, len(rc.Context.Roles))
    for range rc.Context.Roles {
        r := <-ch
        if r.err != nil {
            // 单个角色失败不中断整体流程
            feedback.Opinions[r.index] = &Opinion{
                RoleID:   rc.Context.Roles[r.index].RoleID,
                RoleName: rc.Context.Roles[r.index].Name,
                Content:  "[发言异常]",
            }
            continue
        }
        feedback.Opinions[r.index] = r.op
    }
    return feedback, nil
}
```

```go
type SessionStateMachine struct {
    current State
    transitions map[State]map[Event]State
}

func NewSessionStateMachine() *SessionStateMachine {
    return &SessionStateMachine{
        current: StateDraft,
        transitions: map[State]map[Event]State{
            StateDraft: {
                EventStart: StateRunning,
            },
            StateRunning: {
                EventPause:      StatePaused,
                EventTerminate:  StateEnded,
                EventComplete:   StateEnded,
                EventTimeout:    StateEnded,
            },
            StatePaused: {
                EventResume: StateRunning,
                EventTerminate: StateEnded,
            },
        },
    }
}

func (m *SessionStateMachine) Transition(event Event) error {
    next, ok := m.transitions[m.current][event]
    if !ok {
        return fmt.Errorf("cannot transition from %s via %s", m.current, event)
    }
    m.current = next
    return nil
}
```

## 8. 错误码字典

| 错误码 | HTTP 状态码 | 说明 | 处理方式 |
|--------|-----------|------|---------|
| **E1001** | 504 | Provider 调用超时 | 重试 3 次，指数退避（1s/2s/4s）；失败后跳过该角色发言 |
| **E1002** | 502 | Provider 返回异常响应 | 记录错误到 audit_logs，跳过该角色本轮 |
| **E1003** | 503 | Provider 服务不可用 | 触发熔断，5 分钟内不路由到该 Provider |
| **E2001** | 400 | Graph 构建参数不合法 | 返回校验错误详情，引导用户修正配置 |
| **E2002** | 409 | Session 状态转换非法 | 返回当前状态+允许的操作列表 |
| **E2003** | 404 | Session 不存在 | 检查 session_id 是否正确 |
| **E2004** | 429 | GraphPool 已满 | 提示稍后重试或升级套餐 |
| **E3001** | 200 | 上下文超长（自动恢复） | 触发 ContextManager 压缩策略 |
| **E3002** | 500 | API Key 解密失败 | 联系管理员重新配置 API Key |
| **E4001** | 504 | Agent 异步执行超时 | 标记任务为 failed，通过 SSE 通知用户 |
| **E4002** | 500 | Agent 分发失败 | 返回任务创建失败，建议手动分配 |
| **E5001** | 503 | 数据库连接失败 | 触发熔断，自动重连；返回友好提示 |
| **E5002** | 503 | Redis 连接失败 | SSE 降级为轮询模式 |
| **E6001** | 401 | JWT 过期 | 返回 refresh_token 提示 |
| **E6002** | 403 | 无权限操作资源 | 返回所需权限级别 |
